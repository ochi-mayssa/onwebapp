# Python example
import requests
 
BASE = "https://your-domain/api/branding"
 
t = requests.post(
    f"{BASE}/auth/token/",
    json={"username": u, "password": p})
 
headers = {
    "Authorization": f"Bearer {t.json()['access']}"
}
 
r = requests.get(
    f"{BASE}/requests/",
    params={"status": "in_progress",
        "page_size": 50},
    headers=headers)
 
print(r.json()["count"])
